"use client";

import { useRef } from "react";
import { useFrame } from "@react-three/fiber";
import { RoundedBox } from "@react-three/drei";
import * as THREE from "three";

const pores = [[-.24, .28, .07], [.23, .29, .055], [-.26, -.22, .045], [.27, -.16, .065], [-.1, .34, .03], [.1, -.3, .04]];

function Shoe({ side, legRef }: { side: number; legRef: React.Ref<THREE.Group> }) {
  return <group ref={legRef} position={[side * .17, .56, 0]}>
    <mesh position={[0, -.1, 0]} castShadow><cylinderGeometry args={[.035, .035, .26, 8]} /><meshStandardMaterial color="#f5ce44" /></mesh>
    <mesh position={[0, -.26, 0]}><cylinderGeometry args={[.045, .045, .12, 8]} /><meshStandardMaterial color="#fff9e9" /></mesh>
    <mesh position={[0, -.24, 0]}><cylinderGeometry args={[.047, .047, .015, 8]} /><meshStandardMaterial color="#d3634a" /></mesh>
    <RoundedBox args={[.19, .12, .29]} radius={.05} position={[side * .025, -.36, .06]} castShadow><meshStandardMaterial color="#51465e" roughness={.7} /></RoundedBox>
  </group>;
}

export function BrainBob({ walking = true, carrying = false, inspector = false, phase = 0 }: {
  walking?: boolean; carrying?: boolean; inspector?: boolean; phase?: number;
}) {
  const body = useRef<THREE.Group>(null);
  const leftLeg = useRef<THREE.Group>(null);
  const rightLeg = useRef<THREE.Group>(null);
  const leftArm = useRef<THREE.Group>(null);
  const rightArm = useRef<THREE.Group>(null);
  useFrame(({ clock }) => {
    const step = clock.elapsedTime * 8 + phase;
    const sway = walking ? Math.sin(step) * .45 : Math.sin(step * .2) * .035;
    if (leftLeg.current) leftLeg.current.rotation.x = sway;
    if (rightLeg.current) rightLeg.current.rotation.x = -sway;
    if (leftArm.current) leftArm.current.rotation.x = -sway * .6;
    if (rightArm.current) rightArm.current.rotation.x = carrying ? -.7 : sway * .6;
    if (body.current) { body.current.position.y = walking ? Math.abs(Math.sin(step)) * .035 : Math.sin(step * .2) * .012; body.current.rotation.z = sway * .04; }
  });
  return <group ref={body}>
    <mesh position={[0, 1.02, 0]} castShadow>
      <sphereGeometry args={[.43, 32, 24]} /><meshStandardMaterial color="#f6cf3b" roughness={.85} />
    </mesh>
    {pores.map(([x, y, size], i) => {
      const z = Math.sqrt(.43 ** 2 - x ** 2 - y ** 2);
      return <mesh key={i} position={[x, 1.02 + y, z + .006]} rotation={[-Math.atan2(y, z), Math.atan2(x, z), 0]} scale={[1, .85, .15]}>
        <sphereGeometry args={[size, 10, 8]} /><meshStandardMaterial color="#c9a32d" />
      </mesh>;
    })}
    {[-1, 1].map((side) => <group key={side}>
      <mesh position={[side * .16, 1.15, .38]} scale={[1, 1.13, .6]}><sphereGeometry args={[.155, 20, 16]} /><meshStandardMaterial color="#fffdf0" /></mesh>
      <mesh position={[side * .155, 1.15, .473]}><sphereGeometry args={[.076, 16, 12]} /><meshStandardMaterial color="#60b6bd" /></mesh>
      <mesh position={[side * .155, 1.15, .53]}><sphereGeometry args={[.042, 14, 12]} /><meshStandardMaterial color="#172c30" /></mesh>
      <mesh position={[side * .155 - .016, 1.177, .56]}><sphereGeometry args={[.014, 8, 8]} /><meshBasicMaterial color="white" /></mesh>
      <mesh position={[side * .27, .96, .34]} scale={[1, .55, .2]}><sphereGeometry args={[.07, 12, 8]} /><meshStandardMaterial color="#ed9c5f" /></mesh>
      <mesh position={[side * .065, .85, .406]}><boxGeometry args={[.075, .105, .055]} /><meshStandardMaterial color="#fffdf1" /></mesh>
      <group ref={side < 0 ? leftArm : rightArm} position={[side * .46, .97, 0]}>
        <mesh><sphereGeometry args={[.09, 12, 8]} /><meshStandardMaterial color="#fff9ed" /></mesh>
        <mesh position={[side * .055, -.18, 0]} rotation={[0, 0, side * .12]}><cylinderGeometry args={[.025, .025, .32, 8]} /><meshStandardMaterial color="#f7ce42" /></mesh>
        <mesh position={[side * .07, -.35, 0]}><sphereGeometry args={[.06, 10, 8]} /><meshStandardMaterial color="#f7ce42" /></mesh>
      </group>
    </group>)}
    <mesh position={[0, 1.035, .47]} scale={[1.15, .85, .9]}><sphereGeometry args={[.065, 12, 10]} /><meshStandardMaterial color="#f6ce3e" /></mesh>
    <mesh position={[0, .89, .414]}><boxGeometry args={[.32, .016, .015]} /><meshStandardMaterial color="#a16c25" /></mesh>
    <mesh position={[0, 1.02, 0]} castShadow>
      <sphereGeometry args={[.435, 32, 16, 0, Math.PI * 2, 1.98, Math.PI - 1.98]} />
      <meshStandardMaterial color="#8b7ab6" />
    </mesh>
    <mesh position={[0, 1.02, 0]}>
      <sphereGeometry args={[.439, 32, 4, 0, Math.PI * 2, 1.98, .1]} />
      <meshStandardMaterial color="#fff9ed" />
    </mesh>
    <RoundedBox args={[.16, .1, .025]} radius={.02} position={[.15, .74, .307]}>
      <meshStandardMaterial color="#b6a7d2" />
    </RoundedBox>
    {/* A tiny two-lobed memory badge replaces the tie and belt. */}
    {[-1, 1].map((side) => <mesh key={`badge-${side}`} position={[.15 + side * .027, .74, .332]} scale={[1, 1.2, .35]}>
      <sphereGeometry args={[.033, 10, 8]} /><meshStandardMaterial color="#fff2bf" />
    </mesh>)}
    <Shoe side={-1} legRef={leftLeg} /><Shoe side={1} legRef={rightLeg} />
    {carrying ? <RoundedBox args={[.3, .38, .04]} radius={.018} position={[.51, .7, .3]} rotation={[-.25, 0, -.25]}><meshStandardMaterial color="#fff5ce" /></RoundedBox> : null}
    {inspector ? <group position={[0, 1.47, 0]}>
      <mesh><cylinderGeometry args={[.32, .35, .1, 16]} /><meshStandardMaterial color="#8b7ab6" /></mesh>
      <mesh position={[0, .07, 0]}><cylinderGeometry args={[.2, .25, .13, 16]} /><meshStandardMaterial color="#8b7ab6" /></mesh>
    </group> : null}
  </group>;
}
